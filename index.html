function copyToClipboard(elementId) { let cleanHtml = '
'; let plainText = ''; if (elementId === 'cleanedNumbers') { const nameToDataMap = new Map(); contacts.forEach(contact => { if (!nameToDataMap.has(contact.name)) { nameToDataMap.set(contact.name, { numbers: [], emails: [], addresses: [] }); } const entry = nameToDataMap.get(contact.name); contact.numbers.forEach(n => { if (!entry.numbers.includes(n)) entry.numbers.push(n); }); [...new Set(contact.emails || [])].forEach(e => { if (!entry.emails.includes(e)) entry.emails.push(e); }); [...new Set(contact.addresses || [])].forEach(a => { if (!entry.addresses.includes(a)) entry.addresses.push(a); }); }); nameToDataMap.forEach(({ numbers, emails, addresses }, name) => { // استخدام بدلاً من span cleanHtml += `
${name}
`; plainText += name + '\n'; addresses.forEach(addr => { cleanHtml += `
📍 ${addr}
`; plainText += addr + '\n'; }); emails.forEach(email => { cleanHtml += `
${email}
`; plainText += email + '\n'; }); numbers.forEach(number => { const formatted = formatPhoneNumber(number); const result = lookupResultsMap.get(number); let colorCode = '#ffffff'; // أبيض افتراضي if (result === 'yes') { colorCode = '#4ade80'; // أخضر } else if (result === 'no') { colorCode = '#ff3b30'; // أحمر } // 👇 هذا هو السطر الحاسم: استخدام بدلاً من span cleanHtml += `
${formatted}
`; plainText += formatted + '\n'; }); cleanHtml += '

'; plainText += '\n'; }); } else { const element = document.getElementById(elementId); plainText = element.innerText; cleanHtml += `

${plainText}

`; } cleanHtml += '
'; const container = document.createElement('div'); container.innerHTML = cleanHtml; container.style.position = 'fixed'; container.style.left = '-9999px'; document.body.appendChild(container); const selection = window.getSelection(); const range = document.createRange(); range.selectNodeContents(container); selection.removeAllRanges(); selection.addRange(range); try { document.execCommand('copy'); showNotification('Copied to clipboard! (Evernote compatible)', 'success'); } catch (err) { // Fallback to plain text const textArea = document.createElement("textarea"); textArea.value = plainText; textArea.style.position = "fixed"; textArea.style.left = "-9999px"; document.body.appendChild(textArea); textArea.focus(); textArea.select(); try { document.execCommand('copy'); showNotification('Copied as plain text.', 'success'); } catch (e) { showNotification('حدث خطأ أثناء النسخ', 'error'); } document.body.removeChild(textArea); } selection.removeAllRanges(); document.body.removeChild(container); }
